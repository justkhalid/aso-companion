     3→/* ---------- tiny ZIP (store, no compression) ---------- */
     4→const CRC_T = (function(){ const t = new Uint32Array(256); for(let n = 0; n < 256; n++){ let c = n; for(let k = 0; k < 8; k++) c = (c & 1) ? (0xEDB88320 ^ (c >>> 1)) : (c >>> 1); t[n] = c; } return t; })();
     5→function crc32(u8){ let c = 0xFFFFFFFF; for(let i = 0; i < u8.length; i++) c = CRC_T[(c ^ u8[i]) & 0xFF] ^ (c >>> 8); return (c ^ 0xFFFFFFFF) >>> 0; }
     6→function zipBuild(files){
     7→  const enc = new TextEncoder(), chunks = [], central = []; let off = 0;
     8→  const u16 = v => [v & 255, (v >>> 8) & 255], u32 = v => [v & 255, (v >>> 8) & 255, (v >>> 16) & 255, (v >>> 24) & 255];
     9→  files.forEach(f => {
    10→    const name = enc.encode(f.name), d = (f.data instanceof Uint8Array) ? f.data : enc.encode(f.data), crc = crc32(d);
    11→    const lh = new Uint8Array([80,75,3,4].concat(u16(20), u16(0), u16(0), u16(0), u16(0), u32(crc), u32(d.length), u32(d.length), u16(name.length), u16(0)));
    12→    chunks.push(lh, name, d);
    13→    central.push({name, crc, size: d.length, off});
    14→    off += lh.length + name.length + d.length;
    15→  });
    16→  const cdStart = off; let cdSize = 0;
    17→  central.forEach(c => {
    18→    const ch = new Uint8Array([80,75,1,2].concat(u16(20), u16(20), u16(0), u16(0), u16(0), u16(0), u32(c.crc), u32(c.size), u32(c.size), u16(c.name.length), u16(0), u16(0), u16(0), u16(0), u32(0), u32(c.off)));
    19→    chunks.push(ch, c.name); cdSize += ch.length + c.name.length;
    20→  });
    21→  chunks.push(new Uint8Array([80,75,5,6].concat(u16(0), u16(0), u16(central.length), u16(central.length), u32(cdSize), u32(cdStart), u16(0))));
    22→  return new Blob(chunks, {type:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet'});
    23→}
    24→function iOSLike(){ /* v2.8: iPhone / iPad - the only place blob downloads silently fail */
    25→  try{
    26→    const ua = navigator.userAgent || '';
    27→    if(/iPad|iPhone|iPod/.test(ua)) return true;
    28→    if(navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1) return true; /* iPadOS 13+ */
    29→  }catch(e){}
    30→  return false;
    31→}
    32→function downloadBlob(blob, name){
    33→  /* v2.8: on iPhone/iPad a programmatic blob anchor click is ignored, which is
    34→     why exports "did nothing" there. Hand the file to the share sheet instead
    35→     (Save Image / Save to Files); everywhere else the normal download runs. */
    36→  try{
    37→    if(iOSLike() && navigator.canShare){
    38→      const f = new File([blob], name, {type: blob.type || 'application/octet-stream'});
    39→      if(navigator.canShare({files: [f]})){
    40→        navigator.share({files: [f]}).catch(() => {});
    41→        return;
    42→      }
    43→    }
    44→  }catch(e){}
    45→  const a = document.createElement('a');
    46→  a.href = URL.createObjectURL(blob); a.download = name; a.rel = 'noopener';
    47→  document.body.appendChild(a); a.click(); a.remove();
    48→  setTimeout(() => URL.revokeObjectURL(a.href), 4000);
    49→}

    72→function buildSheetXml(sheet, pool){
    73→  const cols = sheet.cols.map((w, i) => '<col min="' + (i + 1) + '" max="' + (i + 1) + '" width="' + w + '" customWidth="1"/>').join('');
    74→  let rows = '';
    75→  sheet.rows.forEach((r, ri) => {
    76→    let cells = '';
    77→    (r.cells || []).forEach(c => {
    78→      if(c === null || c === undefined) return;
    79→      const v = c.v;
    80→      if(v === '' || v === null || v === undefined) return;   // skip empties
    81→      const ref = colLetter(c.col) + (ri + 1);
    82→      /* +1: the openpyxl-generated styles.xml has its own default xf at index 0,
    83→         so our style k (0..13) lives at cellXfs index k+1 */
    84→      const st = (c.s == null ? -1 : c.s) + 1;
    85→      if(typeof v === 'number' && isFinite(v)){
    86→        cells += '<c r="' + ref + '" s="' + st + '"><v>' + v + '</v></c>';
    87→      } else {
    88→        const key = String(v);
    89→        if(!pool.has(key)) pool.set(key, pool.size);
    90→        cells += '<c r="' + ref + '" t="s" s="' + st + '"><v>' + pool.get(key) + '</v></c>';
    91→      }
    92→    });
    93→    rows += '<row r="' + (ri + 1) + '"' + (r.h ? ' ht="' + r.h + '" customHeight="1"' : '') + '>' + cells + '</row>';
    94→  });
    95→  let merges = '';
    96→  if(sheet.merges && sheet.merges.length){
    97→    merges = '<mergeCells count="' + sheet.merges.length + '">' + sheet.merges.map(m => '<mergeCell ref="' + m + '"/>').join('') + '</mergeCells>';
    98→  }
    99→  return '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' +
   100→    '<worksheet xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships">' +
   101→    '<sheetViews><sheetView workbookViewId="0" showGridLines="0"><pane ySplit="' + sheet.freeze + '" topLeftCell="A' + (sheet.freeze + 1) + '" activePane="bottomLeft" state="frozen"/></sheetView></sheetViews>' +
   102→    '<sheetFormatPr defaultRowHeight="15"/>' +
   103→    '<cols>' + cols + '</cols>' +
   104→    '<sheetData>' + rows + '</sheetData>' +
   105→    merges +
   106→    '<drawing r:id="rId1"/></worksheet>';
   107→}

   119→function b64ToU8(b64){
   120→  const bin = atob(b64), u8 = new Uint8Array(bin.length);
   121→  for(let i = 0; i < bin.length; i++) u8[i] = bin.charCodeAt(i);
   122→  return u8;
   123→}

   126→function xlsxDownload(sheets, filename){
   127→  const logoB64 = LOGO.split(',')[1];
   128→  const pool = new Map();                        // shared strings (LibreOffice drops inlineStr styles)
   129→  const sheetXmls = sheets.map(s => buildSheetXml(s, pool));
   130→  const sst = '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' +
   131→    '<sst xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" count="' + pool.size + '" uniqueCount="' + pool.size + '">' +
   132→    Array.from(pool.keys()).map(k => '<si><t xml:space="preserve">' + X_ESC(k) + '</t></si>').join('') + '</sst>';
   133→  const files = [
   134→    {name:'[Content_Types].xml', data: CT_XML
   135→      .replace('{SHEETS}', sheets.map((s, i) => '<Override PartName="/xl/worksheets/sheet' + (i + 1) + '.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.worksheet+xml"/>').join(''))
   136→      .replace('{DRAWINGS}', sheets.map((s, i) => '<Override PartName="/xl/drawings/drawing' + (i + 1) + '.xml" ContentType="application/vnd.openxmlformats-officedocument.drawing+xml"/>').join(''))
   137→      .replace('{THEME}', '<Override PartName="/xl/theme/theme1.xml" ContentType="application/vnd.openxmlformats-officedocument.theme+xml"/>')
   138→      .replace('{SST}', '<Override PartName="/xl/sharedStrings.xml" ContentType="application/vnd.openxmlformats-officedocument.spreadsheetml.sharedStrings+xml"/>')},
   139→    {name:'_rels/.rels', data: RELS_ROOT},
   140→    {name:'xl/workbook.xml', data: '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' +
   141→      '<workbook xmlns="http://schemas.openxmlformats.org/spreadsheetml/2006/main" xmlns:r="http://schemas.openxmlformats.org/officeDocument/2006/relationships"><sheets>' +
   142→      sheets.map((s, i) => '<sheet name="' + X_ESC(s.name.slice(0, 31)) + '" sheetId="' + (i + 1) + '" r:id="rId' + (i + 1) + '"/>').join('') +
   143→      '</sheets></workbook>'},
   144→    {name:'xl/_rels/workbook.xml.rels', data: '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' +
   145→      '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">' +
   146→      sheets.map((s, i) => '<Relationship Id="rId' + (i + 1) + '" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/worksheet" Target="worksheets/sheet' + (i + 1) + '.xml"/>').join('') +
   147→      '<Relationship Id="rId' + (sheets.length + 1) + '" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/sharedStrings" Target="sharedStrings.xml"/>' +
   148→      '<Relationship Id="rId' + (sheets.length + 2) + '" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/styles" Target="styles.xml"/>' +
   149→      '<Relationship Id="rId' + (sheets.length + 3) + '" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/theme" Target="theme/theme1.xml"/></Relationships>'},
   150→    {name:'xl/styles.xml', data: STYLES_XML},
   151→    {name:'xl/theme/theme1.xml', data: THEME_XML},
   152→    {name:'xl/sharedStrings.xml', data: sst},
   153→    {name:'xl/media/image1.png', data: b64ToU8(logoB64)},
   154→  ];
   155→  sheets.forEach((s, i) => {
   156→    files.push({name:'xl/worksheets/sheet' + (i + 1) + '.xml', data: sheetXmls[i]});
   157→    files.push({name:'xl/worksheets/_rels/sheet' + (i + 1) + '.xml.rels', data: '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' +
   158→      '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">' +
   159→      '<Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/drawing" Target="../drawings/drawing' + (i + 1) + '.xml"/></Relationships>'});
   160→    files.push({name:'xl/drawings/drawing' + (i + 1) + '.xml', data: buildDrawingXml()});
   161→    files.push({name:'xl/drawings/_rels/drawing' + (i + 1) + '.xml.rels', data: '<?xml version="1.0" encoding="UTF-8" standalone="yes"?>\n' +
   162→      '<Relationships xmlns="http://schemas.openxmlformats.org/package/2006/relationships">' +
   163→      '<Relationship Id="rId1" Type="http://schemas.openxmlformats.org/officeDocument/2006/relationships/image" Target="../media/image1.png"/></Relationships>'});
   164→  });
   165→  downloadBlob(zipBuild(files), filename);
   166→}

   169→function schemeSheetRows(lv){
   170→  const n1 = s1Weeks();
   171→  const HDR = ['Week', 'Dates', 'Theme', 'Objectives — by the end, students can…', 'Key language', 'Skills — L / S / R / W', 'Resources (Drive library)', 'Activities', 'Homework', 'Note'];
   172→  const COLS = [2.5, 7, 12, 30, 40, 28, 26, 30, 32, 24, 22];
   173→  const rows = [];
   174→  const C = (col, v, s) => ({col, v, s});
   175→  rows.push({cells: [C(1, ''), C(4, 'ELTASO — ' + lv.label + ' · Schemes of Work', 1)], h: 26});
   176→  rows.push({cells: [C(1, ''), C(4, state.settings.institute + ' · ' + state.settings.year + ' · ' + lv.weeks.length + ' weeks · ' + (lv.cefr || '') + ' · exported from ELTASO Companion v' + APP_VERSION, 2)], h: 16});
   177→  rows.push({cells: [], h: 6});
   178→  rows.push({cells: [], h: 40});
   179→  rows.push({cells: HDR.map((t, i) => C(i + 2, t, 3)), h: 30});
   180→  lv.weeks.forEach((w, wi) => {
   181→    const zebra = wi % 2 === 1;
   182→    const assess = /assessment/i.test(w.theme || '') || /ASSESSMENT/i.test(noteFor(wi));
   183→    const sBody = assess ? 8 : (zebra ? 5 : 4);
   184→    const sWk = zebra ? 7 : 6;
   185→    const sDate = zebra ? 11 : 10;
   186→    const sNote = assess ? 12 : (zebra ? 13 : 12);
   187→    const d = fmtD(weekMonday(wi), {day:'numeric', month:'short'});
   188→    rows.push({cells: [
   189→      C(1, ''),
   190→      C(2, wi + 1, sWk),
   191→      C(3, d, sDate),
   192→      C(4, w.theme || '', assess ? 8 : (zebra ? 5 : 4)),
   193→      C(5, w.obj || '', sBody),
   194→      C(6, w.lang || '', sBody),
   195→      C(7, skillsText(w), sBody),
   196→      C(8, w.res || '', sBody),
   197→      C(9, w.act || '', sBody),
   198→      C(10, w.hw || '', sBody),
   199→      C(11, noteFor(wi), sNote),
   200→    ]});
   201→  });
   202→  return {name: lv.label, rows, cols: COLS, merges: ['D1:K1', 'D2:K2'], freeze: 5};
   203→}

   370→function gridDataFrom(list, isClubs){
   371→  const slots = [], seen = {};
   372→  list.forEach(c => { const s = parseSlot(c.time); if(!seen[s.k]){ seen[s.k] = 1; slots.push(s); } });
   373→  slots.sort((a, b) => a.mins - b.mins || a.label.localeCompare(b.label));
   374→  const cells = {};
   375→  slots.forEach(sl => { cells[sl.k] = {}; DAY_KEYS.forEach(d => cells[sl.k][d] = []); });
   376→  const counters = {};
   377→  let maxChips = 1;

   379→    const s = parseSlot(c.time);
   380→    (c.days || []).forEach(d => {
   381→      let top, main, meta, tone, icon;

   386→        tone = clubToneIdx(c.id);
   387→        icon = clubIconKind(c); /* v2.7: theme icon rides along, drawn when themed */

   390→        top = c.level + ' \u00b7 ' + counters[c.level];
   391→        main = [c.teacher, c.room].filter(Boolean).join(' \u00b7 ');
   392→        meta = '';

   401→function rrPath(ctx, x, y, w, h, r){
   402→  ctx.beginPath();
   403→  ctx.moveTo(x + r, y);
   404→  ctx.arcTo(x + w, y, x + w, y + h, r);
   405→  ctx.arcTo(x + w, y + h, x, y + h, r);
   406→  ctx.arcTo(x, y + h, x, y, r);
   407→  ctx.arcTo(x, y, x + w, y, r);
   408→  ctx.closePath();
   409→}
   410→function clipTo(ctx, t, w){
   411→  const s = String(t == null ? '' : t);

   414→  return s + '\u2026';

   418→function clubIconKind(c){
   419→  const s = ((c.name || '') + ' ' + (c.desc || '')).toLowerCase();
   420→  const has = a => a.some(k => s.indexOf(k) > -1);
   421→  if(has(['chess'])) return 'chess';
   422→  if(has(['code', 'coding', 'robot', 'tech', 'stem', 'science', 'math'])) return 'science';
   423→  if(has(['book', 'read', 'story', 'literat', 'poetry'])) return 'book';
   424→  if(has(['movie', 'film', 'cinema'])) return 'film';
   425→  if(has(['debate', 'talk', 'conversation', 'speech', 'toast'])) return 'mic';
   426→  if(has(['art', 'draw', 'paint', 'design', 'craft'])) return 'art';
   427→  if(has(['music', 'song', 'sing', 'guitar', 'choir'])) return 'music';
   428→  if(has(['writ', 'journal'])) return 'pen';
   429→  if(has(['game', 'gam', 'esport'])) return 'game';
   430→  if(has(['homework', 'tutor', 'study', 'help'])) return 'study';
   431→  return 'star';

   659→      : (state.clubs || []).map(c => ({tone: clubToneIdx(c.id), label: c.name})),
